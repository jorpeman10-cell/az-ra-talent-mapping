**契约（逐字段照抄黄金卡，camelCase）：**

```ts
// candidate_assessment cards — camel wire contract certified by federation golden
// fixtures (candidate_{intake,assess,narrative}_*.json). Nullability matters:
// no-band assess cards carry band.p25/p50/p75 = null, breakevenWan = null,
// npvQuickAtTarget = null (z.number().nullable().optional()).
const CandidateAssessmentErrorFields = {
  error: z.string().min(1).max(200).optional(),
  candidates: z.array(z.object({ cid: z.string(), name: z.string() }).strict()).max(32).optional(),
};

const CandidateIntakeResultCardSchema = z.object({
  kind: z.literal('candidate_intake_result'),
  schemaVersion: z.literal('candidate-intake-card-v1'),
  cid: z.string(),
  name: z.string().optional(),
  status: z.string().optional(),          // lifecycle: CREATED/SELF_DONE/HR_DONE/ASSESSED
  selfUrl: z.string().max(500).optional(),
  hrUrl: z.string().max(500).optional(),
  expiresAt: z.string().optional(),
  ...CandidateAssessmentErrorFields,
}).strict();

const CandidateAssessBandSchema = z.object({
  p25: z.number().nullable().optional(),
  p50: z.number().nullable().optional(),
  p75: z.number().nullable().optional(),
  source: z.string(),                     // blend|internal|interpolated|no_internal_data|unavailable
  confidence: z.string(),                 // MEDIUM|LOW
}).strict();

const CandidateAssessResultCardSchema = z.object({
  kind: z.literal('candidate_assess_result'),
  schemaVersion: z.literal('candidate-assess-card-v1'),
  cid: z.string().optional(),
  name: z.string().optional(),
  status: z.string().optional(),          // engine: OK/REJECT_REVIEW
  grade: z.string().nullable().optional(),
  baseGrade: z.string().nullable().optional(),
  effectiveBillingWan: z.number().nullable().optional(),
  verifyCoefficient: z.number().nullable().optional(),
  band: CandidateAssessBandSchema.optional(),
  breakevenWan: z.number().nullable().optional(),
  npvQuickAtTarget: z.number().nullable().optional(),
  topLine: z.string().nullable().optional(),
  topOverlapPct: z.number().nullable().optional(),
  divergent: z.array(z.string()).optional(),
  highDivergence: z.boolean().optional(),
  rationale: z.array(z.string().max(500)).optional(),
  warnings: z.array(z.string().max(500)).optional(),
  ...CandidateAssessmentErrorFields,
}).strict();

const CandidateNarrativeResultCardSchema = z.object({
  kind: z.literal('candidate_narrative_result'),
  schemaVersion: z.literal('candidate-narrative-card-v1'),
  cid: z.string().optional(),
  name: z.string().optional(),
  status: z.string().optional(),          // lifecycle
  grade: z.string().nullable().optional(),
  confidence: z.string().optional(),
  topLine: z.string().nullable().optional(),
  divergent: z.array(z.string()).optional(),
  highDivergence: z.boolean().optional(),
  openNotes: z.array(z.object({
    role: z.enum(['self', 'hr']),
    dimension: z.string(),
    question: z.string().max(300),
    answer: z.string().max(500),
  }).strict()).max(12).optional(),
  ...CandidateAssessmentErrorFields,
}).strict();

const CandidateAssessmentResultCardSchema = z.union([
  CandidateIntakeResultCardSchema,
  CandidateAssessResultCardSchema,
  CandidateNarrativeResultCardSchema,
]);
```
